// --- Flashcard Components ---
const FlashcardMode: React.FC<{ topic: Topic; onBack: () => void }> = ({ topic, onBack }) => {
  const [currentIndex, setCurrentIndex] = useState(0);
  const [isFlipped, setIsFlipped] = useState(false);
  const cards = topic.flashcards;

  const nextCard = () => { setIsFlipped(false); setTimeout(() => setCurrentIndex((i) => (i + 1) % cards.length), 200); };
  const prevCard = () => { setIsFlipped(false); setTimeout(() => setCurrentIndex((i) => (i - 1 + cards.length) % cards.length), 200); };

  return (
    <div className="max-w-2xl mx-auto px-6 py-8">
      {/* Top Nav */}
      <div className="flex items-center justify-between mb-8">
        <button onClick={onBack} className="text-slate-500 hover:text-slate-800 flex items-center gap-1 text-sm font-medium transition">
          <ChevronLeft size={16} /> Back to Chapters
        </button>
        <span className="text-slate-400 text-sm font-mono">{currentIndex + 1} / {cards.length}</span>
      </div>
      
      <div className="text-center mb-8">
        <h2 className="text-sm font-bold text-indigo-600 uppercase tracking-wider">{topic.title}</h2>
        <h1 className="text-2xl font-bold text-slate-900">{topic.subtitle}</h1>
      </div>

      {/* The Card */}
      <div 
        className="group perspective-1000 w-full h-80 cursor-pointer"
        onClick={() => setIsFlipped(!isFlipped)}
      >
        <div className={`relative w-full h-full duration-500 transform-style-3d transition-all ${isFlipped ? 'rotate-y-180' : ''}`}>
          {/* Front */}
          <div className="absolute w-full h-full bg-white rounded-3xl border-2 border-slate-100 shadow-xl shadow-slate-200/50 p-8 flex flex-col items-center justify-center backface-hidden group-hover:border-indigo-100 transition-colors">

// --- Quiz/Test Component ---
const QuizMode: React.FC<{ topic: Topic; onBack: () => void; isTestMode?: boolean }> = ({ topic, onBack, isTestMode = false }) => {
  const [currentQIndex, setCurrentQIndex] = useState(0);
  const [selectedOption, setSelectedOption] = useState<number | null>(null);
  const [isAnswered, setIsAnswered] = useState(false);
  const [score, setScore] = useState(0);
  const [showResults, setShowResults] = useState(false);

  const question = topic.questions[currentQIndex];
  const totalQ = topic.questions.length;

  const handleOptionClick = (index: number) => {
    if (isAnswered && !isTestMode) return;
    setSelectedOption(index);
  };

  const handleSubmitAnswer = () => {
    if (selectedOption === question.correctIndex) setScore(s => s + 1);
    if (!isTestMode) { setIsAnswered(true); } else { handleNext(); }
  };

  const handleNext = () => {
    if (currentQIndex < totalQ - 1) {
      setCurrentQIndex(p => p + 1); setSelectedOption(null); setIsAnswered(false);
    } else {
      setShowResults(true);
    }
  };

  if (showResults) {
    const percentage = Math.round((score / totalQ) * 100);
    return (
      <div className="max-w-md mx-auto mt-12 p-8 bg-white rounded-3xl border border-slate-100 shadow-2xl shadow-slate-200/50 text-center">
        <div className="w-20 h-20 bg-yellow-100 rounded-full flex items-center justify-center mx-auto mb-6">
          <Trophy className="text-yellow-600" size={40} />
        </div>
        <h2 className="text-2xl font-bold text-slate-900 mb-2">{isTestMode ? 'Test Completed!' : 'Quiz Completed!'}</h2>
        <p className="text-slate-500 mb-8">You have finished {topic.subtitle}.</p>
        
        <div className="flex justify-center items-end gap-2 mb-8">
          <span className="text-6xl font-black text-indigo-600">{percentage}</span>
          <span className="text-xl text-indigo-600/60 font-bold mb-2">%</span>
        </div>
        <p className="text-slate-600 font-medium mb-8 bg-slate-50 py-3 rounded-xl">Score: {score} / {totalQ} correct</p>

        <button onClick={onBack} className="w-full py-3.5 bg-slate-900 text-white rounded-xl font-bold hover:bg-slate-800 transition">
          Return to Home
        </button>
      </div>
    );
  }

  return (
    <div className="max-w-2xl mx-auto px-6 py-8">
       <button onClick={onBack} className="text-slate-500 hover:text-slate-800 flex items-center gap-1 mb-8 text-sm font-medium transition">
        <X size={16} /> Quit {isTestMode ? 'Test' : 'Quiz'}
      </button>

      {/* Modern Progress Bar */}
      <div className="flex items-center gap-4 mb-8">
        <div className="flex-1 h-2 bg-slate-100 rounded-full overflow-hidden">
          <div className="h-full bg-indigo-600 transition-all duration-500 ease-out" style={{ width: `${((currentQIndex + 1) / totalQ) * 100}%` }}></div>
        </div>
        <span className="text-xs font-bold text-slate-400 min-w-[3rem]">Q {currentQIndex + 1} / {totalQ}</span>
      </div>

      <div className="mb-8">
        <h3 className="text-xl font-bold text-slate-900 leading-tight">{question.question}</h3>
      </div>

      <div className="space-y-3 mb-8">
        {question.options.map((option, idx) => {
          let stateStyles = "border-slate-200 hover:border-indigo-300 hover:bg-indigo-50/30 text-slate-700";
          if (isAnswered && !isTestMode) {
             if (idx === question.correctIndex) stateStyles = "border-emerald-500 bg-emerald-50 text-emerald-900 font-medium";
             else if (idx === selectedOption) stateStyles = "border-red-500 bg-red-50 text-red-900";
             else stateStyles = "border-slate-100 opacity-50";
          } else if (selectedOption === idx) {
             stateStyles = "border-indigo-600 bg-indigo-50 text-indigo-900 font-medium shadow-sm";
          }

          return (
            <button 
              key={idx}
              onClick={() => handleOptionClick(idx)}
              disabled={isAnswered && !isTestMode}
              className={`w-full p-4 text-left rounded-xl border-2 transition-all duration-200 flex justify-between items-center ${stateStyles}`}
            >
              <span>{option}</span>
              {isAnswered && !isTestMode && idx === question.correctIndex && <Check size={20} className="text-emerald-600"/>}
              {isAnswered && !isTestMode && idx === selectedOption && idx !== question.correctIndex && <X size={20} className="text-red-500"/>}
            </button>
          );
        })}
      </div>

      {/* Feedback or Next Button */}
      <div className="min-h-[6rem]">
        {isAnswered && !isTestMode && (
          <div className={`p-4 rounded-xl mb-4 animate-in fade-in slide-in-from-top-2 ${selectedOption === question.correctIndex ? 'bg-emerald-50 text-emerald-900' : 'bg-red-50 text-red-900'}`}>
            <p className="font-bold flex items-center gap-2">
              {selectedOption === question.correctIndex ? 'Correct!' : 'Incorrect'}
            </p>
            <p className="mt-1 text-sm opacity-90 leading-relaxed">{question.explanation}</p>
          </div>
        )}

        {(!isAnswered && !isTestMode) ? (
          <button 
            onClick={handleSubmitAnswer}
            disabled={selectedOption === null}
            className="w-full py-3.5 bg-indigo-600 text-white rounded-xl font-bold hover:bg-indigo-700 transition disabled:opacity-50 disabled:cursor-not-allowed"
          >
            Submit Answer
          </button>
        ) : (
          <button 
            onClick={isTestMode ? handleSubmitAnswer : handleNext}
            disabled={isTestMode && selectedOption === null}
            className="w-full py-3.5 bg-slate-900 text-white rounded-xl font-bold hover:bg-slate-800 transition flex items-center justify-center gap-2"
          >
            {currentQIndex < totalQ - 1 ? 'Next Question' : 'View Results'} <ChevronRight size={18} />
          </button>
        )}
      </div>
    </div>
  );
};

// ROOT APP
// =========================================

function App() {
  const [currentPage, setCurrentPage] = useState<PageRoute>('home');
  const [selectedTopic, setSelectedTopic] = useState<Topic | null>(null);
  const [studyMode, setStudyMode] = useState<StudyModeType>(null);

  const startStudy = (topic: Topic, mode: StudyModeType) => {
    setSelectedTopic(topic); setStudyMode(mode); setCurrentPage('study');
  };

  const goHome = () => {
    setCurrentPage('home'); setSelectedTopic(null); setStudyMode(null);
  };

  return (
    <div className="min-h-screen bg-[#FAFAFA] font-sans text-slate-600 selection:bg-indigo-100 selection:text-indigo-900">
      {/* Essential CSS for 3D card flips */}
      <style>{`
        .perspective-1000 { perspective: 1000px; }
        .transform-style-3d { transform-style: preserve-3d; }
        .backface-hidden { backface-visibility: hidden; }
        .rotate-y-180 { transform: rotateY(180deg); }
      `}</style>

      <Header currentPage={currentPage} onNavigate={setCurrentPage} />

      {currentPage === 'home' && <HomePage onStart={startStudy} />}
      {currentPage === 'dashboard' && <DashboardPage />}
      
      {currentPage === 'game' && (
        <main className="max-w-5xl mx-auto px-6 py-24 text-center">
          <div className="inline-block p-6 bg-indigo-50 rounded-full text-indigo-500 mb-6">
            <Gamepad2 size={48} />
          </div>
          <h1 className="text-3xl font-bold text-slate-900 mb-4">Arcade Zone</h1>
          <p className="text-slate-500 text-lg mb-8 max-w-md mx-auto">History memory matching games are currently under development. Check back soon!</p>
          <button onClick={goHome} className="px-8 py-3 bg-slate-900 text-white rounded-xl font-bold hover:bg-slate-800 transition">Back Home</button>
        </main>
      )}

      {currentPage === 'study' && selectedTopic && (
        studyMode === 'flashcard' ? <FlashcardMode topic={selectedTopic} onBack={goHome} /> :
        studyMode === 'quiz' ? <QuizMode topic={selectedTopic} onBack={goHome} isTestMode={false} /> :
        studyMode === 'test' ? <QuizMode topic={selectedTopic} onBack={goHome} isTestMode={true} /> : null
      )}
    </div>
  );
}

export default App;
